<script setup>
import { ref, computed, onMounted } from 'vue'

import { ElMessage, ElMessageBox } from 'element-plus'
import { 
    EditPen, Delete, Check, Edit, Loading 
} from '@element-plus/icons-vue'
import { generateExercises } from '@/api/ai'
import { addExerciseQuestions } from '@/api/exercise'

import { marked } from 'marked'  // 新增导入


const generating = ref(false)
const showPreview = ref(false)

// 生成表单
const generateForm = ref({
    sectionId: '',
    questionTypes: ['choice'],
    difficulty: 'medium',
    count: 10,
    knowledgePoints: [],
    requirements: ''
})



// 生成的题目
const generatedQuestions = ref([])

// 计算属性
const selectedQuestions = computed(() => {
    return generatedQuestions.value.filter(q => q.selected)
})



const getDifficultyText = (difficulty) => {
    const map = {
        easy: '简单',
        medium: '中等',
        hard: '困难',
        mixed: '混合难度'
    }
    return map[difficulty] || difficulty
}

const getQuestionTypeText = (type) => {
    const map = {
        choice: '选择题',
        fill: '填空题',
        judge: '判断题',
        short: '简答题',
        programming: '编程题'
    }
    return map[type] || type
}

const getQuestionTypeColor = (type) => {
    const map = {
        choice: 'primary',
        fill: 'success',
        judge: 'warning',
        short: 'info',
        programming: 'danger'
    }
    return map[type] || 'info'
}

const generateQuestions = async () => {
    if (!generateForm.value.questionTypes.length) {
        ElMessage.warning('请选择题目类型')
        return
    }

    generating.value = true
    try {       
        // 调用 AI 生成接口
        const result = await generateExercises({
            courseId: 1, // 这里需要传入实际的课程ID，可以从路由参数或父组件获取
            chapterId: null,
            questionTypes: generateForm.value.questionTypes,
            difficulty: generateForm.value.difficulty,
            count: generateForm.value.count,
            knowledgePoints: generateForm.value.knowledgePoints,
            requirements: generateForm.value.requirements,
            courseName: '当前课程名称', // 从父组件或路由获取
        })

        if (result.success) {
            generatedQuestions.value = result.questions
            ElMessage.success(`成功生成 ${result.questions.length} 道题目`)
        } else {
            ElMessage.error(`生成失败: ${result.error}`)
        }
    } catch (error) {
        ElMessage.error('生成题目失败')
        console.error(error)
    } finally {
        generating.value = false
    }
}

// 新增保存方法
const saveQuestions = async () => {
    if (!generatedQuestions.value.length) {
        ElMessage.warning('没有可保存的题目')
        return
    }
    
    const selected = generatedQuestions.value.filter(q => q.selected)
    if (!selected.length) {
        ElMessage.warning('请至少选择一道题目')
        return
    }
    
    try {
        // 转换数据格式以匹配后端 DTO
        const questionList = selected.map(q => {
            // 类型映射
            const typeMap = {
                'choice': 'single_choice',
                'fill': 'fill_blank',
                'judge': 'true_false',
                'short': 'essay',
                'programming': 'programming'
            }
            
            // 构建选项
            let options = []
            if (q.type === 'choice' && q.options && q.options.length > 0) {
                const optionKeys = ['A', 'B', 'C', 'D', 'E', 'F']
                options = q.options.map((text, index) => ({
                    optionKey: optionKeys[index] || String.fromCharCode(65 + index),
                    optionText: text,
                    isCorrect: q.answer && q.answer.includes(optionKeys[index] || String.fromCharCode(65 + index))
                }))
            } else if (q.type === 'judge') {
                options = [
                    { optionKey: 'A', optionText: '正确', isCorrect: q.answer === '正确' || q.answer === 'A' },
                    { optionKey: 'B', optionText: '错误', isCorrect: q.answer === '错误' || q.answer === 'B' }
                ]
            }
            
           
            return {
                courseId: 1, // 从父组件或路由获取实际课程ID
                chapterId: null,
                questionType: typeMap[q.type] || 'single_choice',
                question: q.title,
                answer: q.answer,
                analysis: q.explanation || '',
                sourceType: 'ai_generated',
                isActive: true,
                options: options
            }
        })
        
        // 调用批量新增接口
        const response = await addExerciseQuestions(questionList)
        
        if (response.code === '200') {
            ElMessage.success(`成功保存 ${questionList.length} 道题目`)
            // 清空已保存的题目或标记为已保存
            generatedQuestions.value = generatedQuestions.value.filter(q => !q.selected)
        } else {
            ElMessage.error(`保存失败: ${response.message}`)
        }
    } catch (error) {
        ElMessage.error('保存题目失败')
        console.error(error)
    }
}


const clearResult = () => {
    generatedQuestions.value = []
    ElMessage.info('已清空生成结果')
}


const selectAll = () => {
    generatedQuestions.value.forEach(q => q.selected = true)
}

const selectNone = () => {
    generatedQuestions.value.forEach(q => q.selected = false)
}

const editQuestion = (question) => {
    ElMessage.info(`编辑题目：${question.title}`)
    // 这里可以打开编辑弹窗
}

const deleteQuestion = async (question) => {
    try {
        await ElMessageBox.confirm(
            '确定要删除这道题目吗？',
            '确认删除',
            {
                confirmButtonText: '确定',
                cancelButtonText: '取消',
                type: 'warning',
            }
        )
        
        const index = generatedQuestions.value.findIndex(q => q.id === question.id)
        if (index > -1) {
            generatedQuestions.value.splice(index, 1)
            ElMessage.success('删除成功')
        }
    } catch (error) {
        if (error !== 'cancel') {
            ElMessage.error('删除失败')
        }
    }
}





onMounted(() => {
    // 初始化页面
})
</script>
